cmake_minimum_required(VERSION 3.8)
project(rgbd_cloud_demo)
find_package(ament_cmake REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(image_transport REQUIRED)
find_package(message_filters REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core highgui imgproc objdetect)
add_executable(rgbd_cloud src/rgbd_cloud.cpp)
target_compile_features(rgbd_cloud PUBLIC cxx_std_17)
target_compile_options(rgbd_cloud PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(rgbd_cloud rclcpp sensor_msgs image_transport message_filters cv_bridge)
target_include_directories(rgbd_cloud PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(rgbd_cloud ${OpenCV_LIBRARIES})
install(TARGETS rgbd_cloud DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch rviz DESTINATION share/${PROJECT_NAME})
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
ament_package()
